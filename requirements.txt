# Diffusion Fuzzy System (DFS) environment
#
# Tested on: Linux x86_64, Python 3.8.5, NVIDIA driver supporting CUDA 12.1.
#
#   conda create -n fuzzydiffusion python=3.8.5 pip -y
#   conda activate fuzzydiffusion
#   pip install -r requirements.txt
#
# Notes
# -----
# * torch==2.3.0 pulls the CUDA 12.1 runtime wheels (nvidia-*-cu12, triton)
#   automatically on Linux x86_64.
# * lightning==1.9.5 is required by src/: it is the last line that provides
#   `pytorch_lightning.utilities.distributed.rank_zero_only` (imported by LDM
#   and src/ldmodel.py) together with `lightning.pytorch.loggers`.
#   lightning-cloud must stay < 0.6 (0.6.x breaks lightning 1.9.5).
# * latent-diffusion, taming-transformers and CLIP are installed from source.
#   If github.com is unreachable, replace the git URLs with a mirror.
# * Model weights are not Python packages; they are downloaded on first use:
#     - Salesforce/blip-image-captioning-base  (SMC semantic branch)
#     - bert-base-uncased                      (SMC semantic branch)
#     - torchvision VGG16 IMAGENET1K_V1        (SMC feature branch)
# * Removed from the previous export because they conflict with torch 2.3 /
#   numpy 1.24 or are unused: tensorflow, keras, tensorboard-data-server pins,
#   scann, nvidia-*-cu11, mkl/cudatoolkit conda packages.
